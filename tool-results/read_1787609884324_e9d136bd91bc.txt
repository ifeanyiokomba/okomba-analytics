   101→/**
   102→ * Stagger delays are applied via INLINE animation-delay (class-based
   103→ * [animation-delay:*] utilities lose to the unlayered .animate-*
   104→ * shorthands in globals.css) and collapse to 0 under
   105→ * prefers-reduced-motion so every panel settles instantly.
   106→ */
   107→function usePrefersReducedMotion(): boolean {
   108→  return useSyncExternalStore(
   109→    (onChange) => {
   110→      const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   111→      mq.addEventListener("change", onChange);
   112→      return () => mq.removeEventListener("change", onChange);
   113→    },
   114→    () => window.matchMedia("(prefers-reduced-motion: reduce)").matches,
   115→    () => false
   116→  );
   117→}
   118→
   119→/** ms → resolved animation-delay string (0ms under reduced motion). */
   120→function useStagger(): (ms: number) => string {
   121→  const reduced = usePrefersReducedMotion();
   122→  return (ms: number) => `${reduced ? 0 : ms}ms`;
   123→}
