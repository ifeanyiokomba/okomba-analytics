    74→export function AdminPortal({ onExit }: { onExit: () => void }) {
    75→  const [loggedIn, setLoggedIn] = useState(false);
    76→  const [checking, setChecking] = useState(true);

    79→  useEffect(() => {

    85→        setLoggedIn(false);
    86→      } finally {
    87→        setChecking(false);
    88→      }
    89→    })();
    90→  }, []);
    91→
    92→  if (checking) {
    93→    return (

    98→  }

   100→  return loggedIn ? (
   101→    <AdminDashboard onLogout={() => setLoggedIn(false)} onExit={onExit} />
   102→  ) : (
   103→    <AdminLogin onLogin={() => setLoggedIn(true)} onExit={onExit} />

   227→function AdminDashboard({ onLogout, onExit }: { onLogout: () => void; onExit: () => void }) {
   228→  const [inquiries, setInquiries] = useState<Inquiry[]>([]);
   229→  const [subscribers, setSubscribers] = useState<Subscriber[]>([]);
   230→  const [stats, setStats] = useState<Stats | null>(null);
   231→  const [loading, setLoading] = useState(true);
   232→  const [error, setError] = useState<string | null>(null);
   233→  const [updatingId, setUpdatingId] = useState<string | null>(null);
   234→  const [detailService, setDetailService] = useState<Service | null>(null);
   235→  const [detailInquiry, setDetailInquiry] = useState<Inquiry | null>(null);
   236→
   237→  // table controls
   238→  const [search, setSearch] = useState("");
   239→  const [statusFilter, setStatusFilter] = useState<string>("all");
   240→  const [budgetFilter, setBudgetFilter] = useState<string>("all");
   241→  const [page, setPage] = useState(1);
   242→  const [sortKey, setSortKey] = useState<"createdAt" | "name" | "service">("createdAt");
   243→  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");
   244→  const PAGE_SIZE = 10;
   245→
   246→  const toggleSort = (key: "createdAt" | "name" | "service") => {
   247→    if (sortKey === key) {
   248→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   249→    } else {
   250→      setSortKey(key);
   251→      setSortDir(key === "createdAt" ? "desc" : "asc");
   252→    }
   253→  };
   254→
   255→  const load = useCallback(async () => {
   256→    setError(null);
   257→    try {
   258→      const [listRes, statsRes, subsRes] = await Promise.all([
   259→        fetch("/api/inquiries"),
   260→        fetch("/api/inquiries?stats=1"),
   261→        fetch("/api/subscribers"),
   262→      ]);
   263→      if (!listRes.ok || !statsRes.ok) throw new Error("Failed to load data — session may have expired");
   264→      const list = await listRes.json();
   265→      const s = await statsRes.json();
   266→      setInquiries(list.inquiries ?? []);
   267→      setStats(s.stats ?? null);
   268→      // subscribers is non-fatal — only admins with valid session get it
   269→      if (subsRes.ok) {
   270→        const subs = await subsRes.json();
   271→        setSubscribers(subs.subscribers ?? []);
   272→      }
   273→    } catch (err) {
   274→      setError(err instanceof Error ? err.message : "Failed to load");
   275→    } finally {
   276→      setLoading(false);
   277→    }
   278→  }, []);
   279→
   280→  useEffect(() => {
   281→    load();
   282→  }, [load]);
   283→
   284→  // Reset to first page whenever filters change
   285→  useEffect(() => {
   286→    setPage(1);
   287→  }, [search, statusFilter, budgetFilter, sortKey, sortDir]);
   288→
   289→  const updateStatus = async (id: string, status: string) => {
   290→    setUpdatingId(id);
   291→    try {
   292→      const res = await fetch("/api/inquiries", {
   293→        method: "PATCH",
   294→        headers: { "Content-Type": "application/json" },
   295→        body: JSON.stringify({ id, status }),
   296→      });
   297→      if (!res.ok) throw new Error("Update failed");
   298→      await load(); // refresh stats + list from server (single source of truth)
   299→    } catch {
   300→      setError("Could not update status. Please refresh.");
   301→    } finally {
   302→      setUpdatingId(null);
   303→    }
   304→  };
   305→
   306→  const logout = async () => {
   307→    try {
   308→      await fetch("/api/admin/logout", { method: "POST" });
   309→    } catch {
   310→      /* ignore */
   311→    }
   312→    onLogout();
   313→  };
   314→
   315→  const statCards = stats
   316→    ? [
   317→        { label: "Total inquiries", value: stats.total, icon: Inbox, accent: "text-gold", bg: "border-gold/25 bg-gold-dim" },
   318→        { label: "New", value: stats.new, icon: CircleDot, accent: "text-gold", bg: "border-gold/25 bg-gold-dim" },
   319→        { label: "Contacted", value: stats.contacted, icon: Mail, accent: "text-[#5b9eff]", bg: "border-[#5b9eff]/25 bg-[#5b9eff]/10" },
   320→        { label: "In progress", value: stats.in_progress, icon: Clock3, accent: "text-purple-300", bg: "border-purple-400/25 bg-purple-400/10" },
   321→        { label: "Closed", value: stats.closed, icon: CheckCircle2, accent: "text-teal", bg: "border-teal/25 bg-teal-dim" },
   322→        { label: "Last 7 days", value: stats.last7Days, icon: TrendingUp, accent: "text-gold-light", bg: "border-gold-light/25 bg-gold-light/10" },
   323→        { label: "Subscribers", value: stats.subscribers, icon: Users, accent: "text-teal", bg: "border-teal/25 bg-teal-dim" },
   324→      ]
   325→    : [];
   326→
   327→  const maxServiceCount = stats?.byService?.[0]?.count ?? 1;
   328→  const maxBudgetCount = stats?.byBudget?.[0]?.count ?? 1;
   329→
   330→  // Filtered inquiries for the table (client-side search + status + budget filters)
   331→  const filteredInquiries = inquiries
   332→    .filter((i) => {
   333→      const matchesStatus = statusFilter === "all" || i.status === statusFilter;
   334→      const matchesBudget =
   335→        budgetFilter === "all" ||
   336→        (budgetFilter === "none" ? !i.budget : i.budget === budgetFilter);
   337→      if (!matchesStatus || !matchesBudget) return false;
   338→      if (!search.trim()) return true;
   339→      const q = search.trim().toLowerCase();
   340→      return (
   341→        i.name.toLowerCase().includes(q) ||
   342→        i.email.toLowerCase().includes(q) ||
   343→        i.service.toLowerCase().includes(q) ||
   344→        (i.phone ?? "").toLowerCase().includes(q) ||
   345→        (i.whatsapp ?? "").toLowerCase().includes(q) ||
   346→        i.message.toLowerCase().includes(q)
   347→      );
   348→    })
   349→    .sort((a, b) => {
   350→      let cmp = 0;
   351→      if (sortKey === "name") cmp = a.name.localeCompare(b.name);
   352→      else if (sortKey === "service") cmp = a.service.localeCompare(b.service);
   353→      else cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   354→      return sortDir === "asc" ? cmp : -cmp;
   355→    });
   356→
   357→  // Pagination slice
   358→  const totalPages = Math.max(Math.ceil(filteredInquiries.length / PAGE_SIZE), 1);
   359→  const safePage = Math.min(page, totalPages);
   360→  const paginatedInquiries = filteredInquiries.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE);
   361→
   362→  return (
   363→    <div className="section-dark min-h-screen bg-background">
   364→      {/* Admin header */}
   365→      <header className="sticky top-0 z-40 border-b border-white/[0.07] bg-[#05070d]/88 backdrop-blur-xl">
   366→        <div className="container-xl flex h-16 items-center justify-between gap-4">
   367→          <div className="flex items-center gap-3.5">
   368→            <OkombaNavLogo />
   369→            <span className="hidden rounded-full border border-gold/30 bg-gold-dim px-3 py-1 font-mono text-[10px] text-gold sm:inline-block">
   370→              ADMIN
   371→            </span>
   372→          </div>
   373→          <div className="flex items-center gap-1.5 sm:gap-2.5">
   374→            <button
   375→              onClick={load}
   376→              aria-label="Refresh data"
   377→              className="inline-flex items-center gap-2 rounded-lg border border-white/[0.09] bg-white/[0.03] px-2.5 py-2 text-[12.5px] font-medium text-muted-foreground transition-colors hover:border-gold/40 hover:text-gold sm:px-3.5"
   378→            >
   379→              <RefreshCw size={13} aria-hidden="true" /> <span className="hidden sm:inline">Refresh</span>
   380→            </button>
   381→            <button
   382→              onClick={onExit}
   383→              aria-label="Back to website"
   384→              className="inline-flex items-center gap-2 rounded-lg border border-white/[0.09] bg-white/[0.03] px-2.5 py-2 text-[12.5px] font-medium text-muted-foreground transition-colors hover:text-foreground sm:px-3.5"
   385→            >
   386→              <ArrowLeft size={13} aria-hidden="true" /> <span className="hidden sm:inline">Site</span>
   387→            </button>
   388→            <button
   389→              onClick={logout}
   390→              aria-label="Log out"
   391→              className="inline-flex items-center gap-2 rounded-lg border border-red-500/25 bg-red-500/[0.08] px-2.5 py-2 text-[12.5px] font-medium text-red-300 transition-colors hover:bg-red-500/15 sm:px-3.5"
   392→            >
   393→              <LogOut size={13} aria-hidden="true" /> <span className="hidden sm:inline">Logout</span>
   394→            </button>
   395→          </div>
   396→        </div>
   397→      </header>
   398→
   399→      <main className="container-xl py-10">
   400→        <h1 className="font-display text-2xl font-bold text-foreground">Inquiry dashboard</h1>
   401→        <p className="mt-1.5 text-[13.5px] text-muted-foreground">
   402→          Live service requests submitted through the website.
   403→        </p>
   404→
   405→        {error && (
   406→          <p role="alert" className="mt-6 rounded-xl border border-red-500/25 bg-red-500/[0.08] px-4 py-3 text-[13px] text-red-300">
   407→            {error}
   408→          </p>
   409→        )}
   410→
   411→        {/* Stat cards */}
   412→        <div className="mt-8 grid grid-cols-2 gap-3 md:grid-cols-4 lg:grid-cols-7">
   413→          {statCards.map((c) => (
   414→            <div key={c.label} className="surface-card p-5">
   415→              <span className={`inline-flex h-9 w-9 items-center justify-center rounded-xl border ${c.bg} ${c.accent}`}>
   416→                <c.icon size={16} aria-hidden="true" />
   417→              </span>
   418→              <p className="mt-3.5 font-display text-[26px] font-bold leading-none text-foreground">{c.value}</p>

   420→            </div>
   421→          ))}
   422→        </div>
   423→
   424→        {/* Top services */}
   425→        {stats && stats.byService.length > 0 && (

   427→            <h2 className="text-[14.5px] font-semibold text-foreground">Most requested services</h2>

   440→              ))}
   441→            </div>
   442→          </div>
   443→        )}
   444→
   445→        {/* Budget distribution */}
   446→        {stats && stats.byBudget.length > 0 && (
   447→          <div className="surface-card mt-4 p-6">
   448→            <h2 className="text-[14.5px] font-semibold text-foreground">Budget distribution</h2>
   449→            <p className="mt-1 text-[11.5px] text-muted-foreground">From inquiries that shared a budget range</p>
   450→            <div className="mt-5 space-y-3.5">
   451→              {stats.byBudget.map((b) => (
   452→                <div key={b.budget} className="flex items-center gap-4">
   453→                  <span className="w-40 shrink-0 truncate text-[12.5px] text-muted-foreground md:w-56">{b.budget}</span>
